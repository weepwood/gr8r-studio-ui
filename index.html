Get started

Use the source directly, without a build step. Every card above has a Code view you can copy.

HTML / JS
<link rel="stylesheet" href="./src/index.css" />
<div class="gr-ui" data-gr-theme="system">
  <button class="gr-button gr-button--primary">
    Create task
  </button>
</div>
<script type="module">
  import { defineGr8rElements, applyGr8rPreferences } from "./src/index.js";
  defineGr8rElements();   // all 12 elements, idempotent
  applyGr8rPreferences(document.querySelector(".gr-ui"), {
    theme: "system", accent: "indigo", density: "comfortable", motion: "full"
  });
</script>
